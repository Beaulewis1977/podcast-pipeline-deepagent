      const bootAttempted = useRef(false);
      const { status } = useSidecarStore();

      useEffect(() => {
        // ... boot logic updating sidecarStore ...

        // ... beforeunload handler ...

      }

      // --- Connected: render full UI ---

        </AppShell>
      );
    }

        case 'audio': return <AudioSyncView />;
        case 'transcript': return <TranscriptView />;
        case 'branding': return <BrandingView />;
        default: return <IngestionView />;
      }
    }
