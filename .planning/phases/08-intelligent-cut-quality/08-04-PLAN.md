# Phase 8 Plan 04: Render — De-Breathing + Noise-Floor Matching

## Objective

After word-boundary snapping, add two audio quality passes before filtergraph
construction:

1. **De-breathing** (`silero-vad`): extend cut boundaries slightly to swallow
   trailing breath sounds.
2. **Noise-floor matching** (`librosa`): detect RMS level mismatch across a join
   and inject a short FFmpeg gain ramp to smooth the transition.

### Task 2: Create vad.py

**File:** `src/podcast_pipeline/utils/vad.py`

```python
"""Silero-VAD wrapper for breath/non-speech detection at cut boundaries."""
from __future__ import annotations

from pathlib import Path
import torch

_model = None  # module-level singleton; loaded once


def _load_model():
    global _model
    if _model is None:
        from silero_vad import load_silero_vad
        _model = load_silero_vad()
    return _model


def detect_breath_at_boundary(
    audio_path: Path,
    boundary_seconds: float,
    *,
    window_ms: float = 200.0,
    max_extend_ms: float = 150.0,
    sampling_rate: int = 16000,
) -> float:
    """
    Return the number of seconds to extend a cut boundary to swallow
    a trailing breath. Returns 0.0 if no breath detected or VAD unavailable.
    """
    from silero_vad import read_audio, get_speech_timestamps

    try:
        model = _load_model()
        audio = read_audio(str(audio_path), sampling_rate=sampling_rate)
    except Exception:
        return 0.0

    window_s = window_ms / 1000.0
    max_extend_s = max_extend_ms / 1000.0

    start_sample = int(max(0, boundary_seconds - window_s) * sampling_rate)
    end_sample = int(boundary_seconds * sampling_rate)
    segment = audio[start_sample:end_sample]

    if len(segment) < sampling_rate * 0.02:   # < 20ms — too short to analyse
        return 0.0

    timestamps = get_speech_timestamps(
        segment, model, sampling_rate=sampling_rate
    )

    # If the end of the segment is non-speech (breath), extend into it
    segment_end_sample = len(segment)
    if not timestamps or timestamps[-1]["end"] < segment_end_sample:
        last_speech_end = timestamps[-1]["end"] if timestamps else 0
        trailing_non_speech_s = (segment_end_sample - last_speech_end) / sampling_rate
        return min(trailing_non_speech_s, max_extend_s)

    return 0.0
```

### Task 3: Create noise_match.py

**File:** `src/podcast_pipeline/utils/noise_match.py`

```python
"""Librosa RMS noise-floor mismatch detection and FFmpeg gain-ramp builder."""
from __future__ import annotations

import numpy as np


def measure_rms_db(audio_array: np.ndarray) -> float:
    """Measure RMS power of an audio array in dBFS."""
    rms = np.sqrt(np.mean(audio_array ** 2))
    if rms < 1e-9:
        return -120.0
    return 20.0 * np.log10(rms)


def build_gain_ramp_filter(
    delta_db: float,
    ramp_ms: float,
    *,
    segment: str = "right",   # "left" or "right"
) -> str | None:
    """
    Return an FFmpeg `volume` filter expression to apply a linear gain ramp
    of ramp_ms that corrects a delta_db mismatch.
    Returns None if no ramp needed (delta below threshold).
    """
    if abs(delta_db) < 0.01:
        return None
    ramp_s = ramp_ms / 1000.0
    # Linear ramp: start at current level, end at target
    gain_start = 10 ** (-abs(delta_db) / 20) if delta_db > 0 else 1.0
    gain_end = 1.0 if delta_db > 0 else 10 ** (abs(delta_db) / 20)
    return (
        f"volume=enable='between(t,0,{ramp_s:.3f})'"
        f":volume='lerp({gain_start:.4f},{gain_end:.4f},t/{ramp_s:.3f})'"
        f":eval=frame"
    )


def check_noise_floor_mismatch(
    audio_left: Path,
    audio_right: Path,
    *,
    window_ms: float = 100.0,
    threshold_db: float = 3.0,
    ramp_ms: float = 50.0,
    sampling_rate: int = 44100,
) -> str | None:
    """
    Load the tail of audio_left and head of audio_right, compare RMS.
    Return an FFmpeg volume filter string if mismatch > threshold, else None.
    """
    try:
        import librosa
        window_samples = int(window_ms / 1000.0 * sampling_rate)
        y_left, _  = librosa.load(str(audio_left), sr=sampling_rate, mono=True)
        y_right, _ = librosa.load(str(audio_right), sr=sampling_rate, mono=True)
        tail  = y_left[-window_samples:] if len(y_left) >= window_samples else y_left
        head  = y_right[:window_samples] if len(y_right) >= window_samples else y_right
        db_left  = measure_rms_db(tail)
        db_right = measure_rms_db(head)
        delta = db_left - db_right
        if abs(delta) > threshold_db:
            return build_gain_ramp_filter(delta, ramp_ms, segment="right")
    except Exception:
        pass
    return None
```
