**Step A — Create `src/podcast_pipeline/utils/pose_match.py`:**

Implement with lazy cv2 import:

```python
from __future__ import annotations
import structlog
from pathlib import Path

logger = structlog.get_logger(__name__)

def pose_distance(frame_a_path: Path, frame_b_path: Path) -> float:
    """Compute optical flow magnitude between two frames. Returns float('inf') if opencv unavailable."""
    try:
        import cv2
        import numpy as np
    except ImportError:
        return float("inf")

    frame_a = cv2.imread(str(frame_a_path))
    frame_b = cv2.imread(str(frame_b_path))
    if frame_a is None or frame_b is None:
        return float("inf")

    gray_a = cv2.cvtColor(frame_a, cv2.COLOR_BGR2GRAY)
    gray_b = cv2.cvtColor(frame_b, cv2.COLOR_BGR2GRAY)
    flow = cv2.calcOpticalFlowFarneback(
        gray_a, gray_b, None,
        pyr_scale=0.5, levels=3, winsize=15,
        iterations=3, poly_n=5, poly_sigma=1.2, flags=0,
    )
    magnitude, _ = cv2.cartToPolar(flow[..., 0], flow[..., 1])
    return float(magnitude.mean())


def scan_best_frame_pair(
    left_frames: list[Path],
    right_frames: list[Path],
    search_window: int = 6,  # frames at 30fps ~= 200ms
) -> tuple[int, int, float]:
    """Scan for frame pair with minimum pose distance.
    Returns (left_idx, right_idx, distance). Clamps window to available frames."""
    # Clamp search window to min(search_window, len(frames))
    left_n = min(search_window, len(left_frames))
    right_n = min(search_window, len(right_frames))

    best_left, best_right, best_dist = 0, 0, float("inf")

    # Scan left_frames[-left_n:] x right_frames[:right_n]
    for li in range(len(left_frames) - left_n, len(left_frames)):
        for ri in range(right_n):
            dist = pose_distance(left_frames[li], right_frames[ri])
            if dist < best_dist:
                best_left, best_right, best_dist = li, ri, dist

    return best_left, best_right, best_dist
```

- Use RIFE model 4.25 (not 4.26) as default.

```python
from __future__ import annotations
import math
import subprocess
from pathlib import Path
import structlog

logger = structlog.get_logger(__name__)

class RifeBridge:
    """Subprocess wrapper for practical-RIFE frame interpolation."""

    def __init__(self, script_path: str = "") -> None:
        self.script_path = Path(script_path) if script_path else Path("")

    def available(self) -> bool:
        """Check if RIFE inference script exists and is accessible."""
        return self.script_path.is_file()

    @staticmethod
    def frames_to_exp(num_frames: int) -> int:
        """Convert desired frame count to --exp value. 4->2, 8->3, 16->4.
        RIFE generates 2^exp frames. Rounds up to next power of 2."""
        return max(1, math.ceil(math.log2(max(num_frames, 1))))

    def generate(
        self,
        frame_a: Path,
        frame_b: Path,
        output_dir: Path,
        num_frames: int = 4,
    ) -> list[Path]:
        """Generate bridge frames using RIFE. Returns empty list if unavailable or failed."""
        if not self.available():
            logger.warning("rife_not_available", script_path=str(self.script_path))
            return []

        output_dir.mkdir(parents=True, exist_ok=True)
        exp = self.frames_to_exp(num_frames)  # 4 frames -> exp=2 (2^2=4)

        cmd = [
            "python", str(self.script_path),
            "--img", str(frame_a), str(frame_b),
            "--output", str(output_dir),
            "--exp", str(exp),
        ]
        # NOTE: No --cpu flag exists (Correction 3). To force CPU:
        # set CUDA_VISIBLE_DEVICES="" in env
        logger.info("rife_generating", exp=exp, num_frames=num_frames, output_dir=str(output_dir))

        result = subprocess.run(cmd, capture_output=True, text=True, check=False)
        if result.returncode != 0:
            logger.error("rife_failed", returncode=result.returncode, stderr=result.stderr[:500])
            return []

        generated = sorted(output_dir.glob("*.png"))
        logger.info("rife_generated", count=len(generated))
        return generated
```

**Step A — Wire into render.py:**

Add a new content-join quality pass in the render pipeline, AFTER the de-breathing + noise-floor passes (from Plan 04), for content-cut joins only (not filler joins):

1. If `smoothing.pose_match_enabled` and the job has video:
   - For each content-cut join point, extract N frames from the tail of the left segment and head of the right segment using FFmpeg (`ffmpeg -ss {time} -i input.mp4 -frames:v {N} -q:v 2 output/%03d.png`). N = `search_window_ms / (1000 / fps)` frames.
   - Call `scan_best_frame_pair(left_frames, right_frames, search_window=N)` to find the optimal cut point.
   - If the best pose distance > `smoothing.pose_match_rife_threshold` AND `smoothing.rife_enabled`:
     - Create a `RifeBridge(script_path=smoothing.rife_script_path)`.
     - Call `rife_bridge.generate(best_left_frame, best_right_frame, temp_dir, num_frames=smoothing.rife_num_bridge_frames)`.
     - If RIFE returns frames, splice them into the filtergraph between the two segments using FFmpeg concat demuxer.
     - If RIFE fails and `smoothing.rife_fallback_to_xfade` is True, fall back to Phase 7 xfade.
   - If pose distance <= threshold, use the best-matched frames as the cut point (adjust trim times) and apply normal Phase 7 xfade/crossfade.
   - If `pose_match_enabled` is False, skip this pass entirely (existing Phase 7 behavior).
   - If opencv is not installed, log a warning and skip pose matching (existing behavior).

2. Audio-only jobs: auto-skip pose_match (no video frames to compare). Log info message.

3. Clean up temp frame directories after each join is processed.
