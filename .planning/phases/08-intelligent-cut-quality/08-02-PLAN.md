def test_filler_config_triage_model_not_openai():

    config = FillerConfig()
    # Must be gemini-3-flash-lite or claude-haiku-4-5 per spec

        f"Triage model must be a lightweight non-reasoning model, got '{config.llm_triage_model}'"
