**Step A — Create `scripts/smoke_test_gpu_rife.py`:**

A standalone script that verifies the GPU + RIFE stack on real hardware (RTX 5060 Ti). NOT a pytest test — it's a manual/CI script that requires GPU.

```python
#!/usr/bin/env python3
"""GPU smoke test for RIFE frame interpolation on RTX 5060 Ti.

Run: python scripts/smoke_test_gpu_rife.py
Expected: prints "RIFE OK" on success, exits non-zero on failure.

Requirements: torch>=2.7 with cu128, practical-RIFE cloned with model weights.
"""
import sys
import tempfile
from pathlib import Path

def main() -> int:
    # Step 1: Verify torch + CUDA
    try:
        import torch
    except ImportError:
        print("FAIL: torch not installed")
        return 1

    if not torch.cuda.is_available():
        print("FAIL: CUDA not available")
        return 1

    cap = torch.cuda.get_device_capability()
    name = torch.cuda.get_device_name(0)
    cuda_version = torch.version.cuda
    print(f"GPU: {name}, capability: {cap}, CUDA: {cuda_version}")

    # Verify Blackwell sm_120 support (Correction 1)
    if cap[0] < 12:
        print(f"WARNING: Expected compute capability >= 12.0 for RTX 5060 Ti, got {cap}")

    # Step 2: Generate synthetic test frames
    try:
        import cv2
        import numpy as np
    except ImportError:
        print("FAIL: opencv-python-headless not installed")
        return 1

    with tempfile.TemporaryDirectory() as tmpdir:
        tmp = Path(tmpdir)
        frame_a = tmp / "frame_a.png"
        frame_b = tmp / "frame_b.png"
        # Create two slightly different frames (720p)
        a = (np.random.rand(720, 1280, 3) * 255).astype(np.uint8)
        b = (np.random.rand(720, 1280, 3) * 255).astype(np.uint8)
        cv2.imwrite(str(frame_a), a)
        cv2.imwrite(str(frame_b), b)

        # Step 3: Run RIFE
        from podcast_pipeline.utils.rife_bridge import RifeBridge
        from podcast_pipeline.config.settings import SmoothingConfig

        config = SmoothingConfig()
        script_path = config.rife_script_path
        if not script_path:
            # Try common location
            default = Path("/opt/practical-rife/inference_img.py")
            if default.exists():
                script_path = str(default)
            else:
                print("FAIL: RIFE script not found. Set smoothing.rife_script_path or clone to /opt/practical-rife/")
                return 1

        bridge = RifeBridge(script_path=script_path)
        if not bridge.available():
            print(f"FAIL: RIFE script not found at {script_path}")
            return 1

        output_dir = tmp / "rife_out"
        frames = bridge.generate(frame_a, frame_b, output_dir, num_frames=4)

        if not frames:
            print("FAIL: RIFE generated no frames")
            return 1

        print(f"RIFE generated {len(frames)} bridge frames")
        print("RIFE OK")
        return 0

if __name__ == "__main__":
    sys.exit(main())
```

Make it executable: `chmod +x scripts/smoke_test_gpu_rife.py`
